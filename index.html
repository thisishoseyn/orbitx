    // Safe Cross-Browser Rounded Rect Helper (Never crashes if roundRect is missing)

    function drawRoundedRect(ctx, x, y, w, h, r) {

      const radius = Math.min(r, w / 2, h / 2);

      ctx.beginPath();

      ctx.moveTo(x + radius, y);

      ctx.lineTo(x + w - radius, y);

      ctx.quadraticCurveTo(x + w, y, x + w, y + radius);

      ctx.lineTo(x + w, y + h - radius);

      ctx.quadraticCurveTo(x + w, y + h, x + w - radius, y + h);

      ctx.lineTo(x + radius, y + h);

      ctx.quadraticCurveTo(x, y + h, x, y + h - radius);

      ctx.lineTo(x, y + radius);

      ctx.quadraticCurveTo(x, y, x + radius, y);

      ctx.closePath();

    }


    // Dynamic ABI Encoding

    function toHexUint256(num) {

      const n = Math.max(0, Math.floor(Number(num) || 0));

      return BigInt(n).toString(16).padStart(64, '0');

    }


    function toHexString(str) {

      const utf8 = new TextEncoder().encode(String(str || ''));

      const lenHex = toHexUint256(utf8.length);

      let dataHex = '';

      for (let i = 0; i < utf8.length; i++) {

        dataHex += utf8[i].toString(16).padStart(2, '0');

      }

      const paddedLen = Math.max(64, Math.ceil(dataHex.length / 64) * 64);

      return lenHex + dataHex.padEnd(paddedLen, '0');

    }


    function encodeClaimProfileABI(username, orbColor) {

    }

    function spawnWaveRow(w) {

      for (let i = 0; i < game.blocks.length; i++) {

        const b = game.blocks[i];

        b.row += 1;

        b.targetY = TOP_OFFSET + b.row * (BH + GAP);

      }

      for (let i = 0; i < game.blocks.length; i++) {

        const b = game.blocks[i];

        if (b.type !== 'steel' && b.type !== 'plus' && b.targetY + BH >= DANGER_Y) {

          game.state = 'OVER';

        }

      }


      const plusCol = Math.floor(Math.random() * COLS);

      let steelSpawned = false;

      let breakableSpawned = 0;


      for (let c = 0; c < COLS; c++) {

        const bx = PAD + c * (BW + GAP);

        const by = TOP_OFFSET;


        if (c === plusCol) {

          game.blocks.push({

            id: Math.random(), col: c, row: 0, x: bx, y: by, targetY: by,

            w: BW, h: BH, type: 'plus', hp: 1, maxHp: 1, hitScale: 1

          });

          continue;

        }


        if (Math.random() > 0.55 && (c < COLS - 1 || breakableSpawned > 0)) continue;


        const baseHp = Math.max(1, Math.round(w * (0.9 + Math.random() * 0.6)));

        const roll = Math.random();

        let type = 'normal';


        if (w >= 3 && !steelSpawned && roll < 0.1) {

          type = 'steel';

          steelSpawned = true;

        } else if (roll < 0.19) {

          type = 'bomb';

          breakableSpawned++;

        } else if (roll < 0.27) {

          type = 'laser';

          breakableSpawned++;

    function addExplosion(x, y, color, count) {

      for (let i = 0; i < count; i++) {

        const s = 1.5 + Math.random() * 4.5;

        game.particles.push({

          x: x, y: y, vx: Math.cos(a) * s, vy: Math.sin(a) * s,

          alpha: 1, color: color, r: 2 + Math.random() * 2

        });

      }

    }


    function damageBlock(rootBlock, initialDmg) {

      const queue = [{ block: rootBlock, dmg: initialDmg }];

      const visited = {};


      while (queue.length > 0) {

        const item = queue.shift();

        const b = item.block;

        if (!b || b.hp <= 0 || b.type === 'steel') continue;


        b.hp -= item.dmg;

        b.hitScale = 0.86;

        game.score += Math.round(15 * app.gmMult);


        if (b.hp <= 0 && !visited[b.id]) {

          visited[b.id] = true;

          const cx = b.x + b.w / 2;

          const cy = b.y + b.h / 2;

          addExplosion(cx, cy, '#38bdf8', 12);

          game.shockwaves.push({ x: cx, y: cy, r: 6, maxR: 45, alpha: 0.9, color: '#38bdf8' });


          if (b.type === 'bomb') {

            game.shockwaves.push({ x: cx, y: cy, r: 10, maxR: 110, alpha: 1, color: '#f97316' });

            for (let i = 0; i < game.blocks.length; i++) {

              const other = game.blocks[i];

              if (other.hp > 0 && other.type !== 'steel' && other.type !== 'plus') {

                if (Math.hypot((other.x + other.w / 2) - cx, (other.y + other.h / 2) - cy) < 110) {

                  queue.push({ block: other, dmg: 8 });

                }

              }

            }

          } else if (b.type === 'laser') {

            game.shockwaves.push({ x: cx, y: cy, r: 10, maxR: 140, alpha: 1, color: '#eab308' });

            for (let i = 0; i < game.blocks.length; i++) {

              const other = game.blocks[i];

              if (other.hp > 0 && other.type !== 'steel' && other.type !== 'plus') {

                if (Math.abs(other.y - b.y) < 12) {

                  queue.push({ block: other, dmg: 6 });

                }

              }

            }

          }

        }

      }

    }


    function stepOrbPhysics(o, dt) {

      o.x += o.vx * dt;

      o.y += o.vy * dt;


      if (o.x < ORB_R) { o.x = ORB_R; o.vx = Math.abs(o.vx); }

      if (o.x > W - ORB_R) { o.x = W - ORB_R; o.vx = -Math.abs(o.vx); }

      if (o.y < ORB_R) { o.y = ORB_R; o.vy = Math.abs(o.vy); }


      for (let i = game.blocks.length - 1; i >= 0; i--) {

        const b = game.blocks[i];

        if (b.hp <= 0) continue;


        if (b.type === 'plus') {

          const cx = b.x + b.w / 2;

          const cy = b.y + b.h / 2;

          if (Math.hypot(o.x - cx, o.y - cy) < ORB_R + 13) {

            b.hp = 0;

            game.bonusOrbsNext += 1;

            addExplosion(cx, cy, '#10b981', 10);

            game.floatTexts.push({ x: cx, y: cy, txt: '+1 ORB', color: '#10b981', alpha: 1 });

            game.blocks.splice(i, 1);

          }

          continue;

        }


        const cx = Math.max(b.x, Math.min(o.x, b.x + b.w));

        const cy = Math.max(b.y, Math.min(o.y, b.y + b.h));

        if (Math.hypot(o.x - cx, o.y - cy) < ORB_R) {

          const l = (o.x + ORB_R) - b.x;

          const r = (b.x + b.w) - (o.x - ORB_R);

          const t = (o.y + ORB_R) - b.y;

          const bb = (b.y + b.h) - (o.y - ORB_R);

          const min = Math.min(l, r, t, bb);


          if (min === l) { o.x = b.x - ORB_R - 0.6; o.vx = -Math.abs(o.vx); }

          else if (min === r) { o.x = b.x + b.w + ORB_R + 0.6; o.vx = Math.abs(o.vx); }

          else if (min === t) { o.y = b.y - ORB_R - 0.6; o.vy = -Math.abs(o.vy); }

          else { o.y = b.y + b.h + ORB_R + 0.6; o.vy = Math.abs(o.vy); }


          if (Math.abs(o.vy) < 0.55) {

            o.vy = (o.vy >= 0 ? 1 : -1) * 0.65;

          }


          if (b.type === 'steel') {

            b.hitScale = 0.93;

            addExplosion(cx, cy, '#94a3b8', 4);

          } else {

            damageBlock(b, 1);

          }

          break;

        }

      }


      game.blocks = game.blocks.filter(function(b) { return b.hp > 0; });

    }


    function updateLoop() {

      for (let i = 0; i < game.blocks.length; i++) {

        const b = game.blocks[i];

        b.y += (b.targetY - b.y) * 0.22;

        b.hitScale += (1 - b.hitScale) * 0.2;

      }


      if (game.state === 'SHOOT') {

        game.shotFrames++;

        const steps = 8 * game.speedMult;

        const dt = 1 / 8;

        const gravityAssist = game.shotFrames > 960 ? 0.008 : 0;


        for (let s = 0; s < steps; s++) {

          if (game.shootQueue > 0) {

            game.shootTimer -= dt;

            if (game.shootTimer <= 0) {

              spawnOrb();

              game.shootQueue--;

              game.shootTimer = 3.2;

            }

          }

          for (let i = game.orbs.length - 1; i >= 0; i--) {

            const o = game.orbs[i];

            o.vy += gravityAssist;

            if (s % 2 === 0) {

              o.trail.push({ x: o.x, y: o.y });

              if (o.trail.length > 6) o.trail.shift();

            }

            stepOrbPhysics(o, dt);

            if (o.y >= LAUNCH_Y) {

              if (game.nextLaunchX === null) {

                game.nextLaunchX = Math.max(28, Math.min(W - 28, o.x));

              }

              game.orbs.splice(i, 1);

            }

          }

        }


        if (game.shootQueue === 0 && game.orbs.length === 0) {

          finishTurn();

        }

      }


      for (let i = game.particles.length - 1; i >= 0; i--) {

        const p = game.particles[i];

        p.x += p.vx;

        p.y += p.vy;

        p.alpha -= 0.035;

        if (p.alpha <= 0) game.particles.splice(i, 1);

      }


      for (let i = game.shockwaves.length - 1; i >= 0; i--) {

        const sw = game.shockwaves[i];

        sw.r += (sw.maxR - sw.r) * 0.2;

        sw.alpha -= 0.045;

        if (sw.alpha <= 0) game.shockwaves.splice(i, 1);

      }


      for (let i = game.floatTexts.length - 1; i >= 0; i--) {

        const ft = game.floatTexts[i];

        ft.y -= 1;

        ft.alpha -= 0.03;

        if (ft.alpha <= 0) game.floatTexts.splice(i, 1);

      }


      updateHud();

    }


    function finishTurn() {

      document.getElementById('recallBtn').style.display = 'none';

      if (game.nextLaunchX !== null) game.launchX = game.nextLaunchX;


      if (game.bonusOrbsNext > 0) {

        game.totalOrbs += game.bonusOrbsNext;

        game.bonusOrbsNext = 0;

      }


      const breakablesLeft = game.blocks.filter(function(b) {

        return b.type !== 'steel' && b.type !== 'plus';

      }).length;


      if (breakablesLeft === 0) {

        const bonus = Math.round(2500 * app.gmMult);

        game.score += bonus;

        app.cleanSweeps += 1;

        const banner = document.getElementById('sweepBanner');

        banner.textContent = '⚡ CLEAN SWEEP! +' + bonus.toLocaleString() + ' PTS';

        banner.classList.add('show');

        setTimeout(function() { banner.classList.remove('show'); }, 1800);

      }


      game.wave += 1;

      if (game.score > app.bestScore) app.bestScore = game.score;

      if (game.wave > app.bestWave) app.bestWave = game.wave;

      saveState();


      game.state = 'AIM';

      spawnWaveRow(game.wave);

      renderLeaderboards();

    }

    function blockColor(b) {

      if (b.type === 'bomb') return '#f97316';

      if (b.type === 'laser') return '#eab308';

      const r = b.hp / Math.max(1, b.maxHp);

      if (r > 0.66) return '#8b5cf6';

      if (r > 0.33) return '#0ea5e9';

      return '#10b981';

    }
